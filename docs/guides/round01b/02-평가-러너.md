# round01b 교본 3-2 — 평가 러너(`eval`)

> [00-개요.md](00-개요.md)의 「3. 구현」 둘째 묶음. 전체 코드 수록, 발췌·플레이스홀더 금지.

`eval`은 이번 라운드에 새로 생긴 워크스페이스 멤버다(`hoa-eval`). 에이전트를 import하지 않고 `hoa investigate --json`을 하위 프로세스로 부른다. 만드는 순서는 러너의 입력부터다 — 시나리오 YAML과 그 모델(`scenario.py`), 에이전트 실행과 받는 쪽 JSON 모양(`agent.py`) → 클러스터 조작(`cluster.py`) → 채점(`scoring.py`) → 러너(`runner.py`)와 결과 표(`report.py`) → CLI(`cli.py`). 단위 테스트는 가짜 kubectl·가짜 시계·가짜 에이전트, 그리고 진짜 `hoa`와 같은 약속(UTF-8 표준출력·종료 코드)을 지키는 짧은 Python 명령을 쓴다. 통합 테스트만 kind에 장애를 실제로 넣는다.

마지막의 `test_kind_cluster.py`는 round01a의 MCP 통합 테스트로, 이번에 장애 대상을 시나리오 YAML에서 읽도록 바뀌었다(장애 대상의 정본을 한 곳으로).

```python
from collections.abc import Sequence
from pathlib import Path
from typing import Literal, Self

import yaml
from pydantic import BaseModel, ConfigDict, Field, ValidationError, model_validator

REPO_ROOT = Path(__file__).resolve().parents[3]
SCENARIO_DIR = REPO_ROOT / "eval" / "scenarios"
FAULTS_DIR = REPO_ROOT / "deploy" / "kind" / "faults"
DEMO_NAMESPACE = "demo"

# hoa_agent.diagnosis.CauseLabel의 값 — 멤버 경계 때문에 import하지 않고 옮겨 둔다
CauseLabel = Literal[
    "image_pull_error",
    "bad_config",
    "oom_killed",
    "probe_failure",
    "service_selector_mismatch",
    "downstream_unavailable",
    "unknown",
]


class ScenarioError(Exception):
    pass


class Fault(BaseModel):
    """장애 패치는 기준 매니페스트에 이미 있는 필드의 값만 바꾼다.

    복구는 기준 매니페스트를 client-side apply로 다시 적용한다 — 패치가 새로 더한 필드는 남는다.
    """

    model_config = ConfigDict(extra="forbid")

    patch: Path
    # 러너는 관리자 컨텍스트로 쓴다 — 코딩 표준이 허용한 데모 네임스페이스 밖은 받지 않는다
    namespace: Literal["demo"]
    deployment: str


class Symptom(BaseModel):
    model_config = ConfigDict(extra="forbid")

    waiting_reasons: list[str] = Field(default_factory=list)
    min_restarts: int | None = Field(default=None, ge=1)
    timeout_seconds: int = Field(default=120, gt=0)

    @model_validator(mode="after")
    def needs_a_condition(self) -> Self:
        if not self.waiting_reasons and self.min_restarts is None:
            raise ValueError("wait needs waiting_reasons or min_restarts")
        return self


class Expected(BaseModel):
    model_config = ConfigDict(extra="forbid")

    cause_label: CauseLabel
    target: str


class Scenario(BaseModel):
    model_config = ConfigDict(extra="forbid")

    name: str
    fault: Fault
    wait: Symptom
    alert: str
    expected: Expected


def load_scenario(path: Path) -> Scenario:
    try:
        scenario = Scenario.model_validate(yaml.safe_load(path.read_text(encoding="utf-8")))
    except (yaml.YAMLError, ValidationError) as exc:
        raise ScenarioError(f"{path.name}: {exc}") from exc
    if scenario.name != path.stem:
        raise ScenarioError(f"{path.name}: name must match the file name, got {scenario.name}")
    patch = (REPO_ROOT / scenario.fault.patch).resolve()
    if FAULTS_DIR.resolve() not in patch.parents or not patch.is_file():
        raise ScenarioError(
            f"{path.name}: patch must be a file under deploy/kind/faults/: {scenario.fault.patch}"
        )
    fault_target = f"{scenario.fault.namespace}/{scenario.fault.deployment}"
    if scenario.expected.target != fault_target:
        raise ScenarioError(
            f"{path.name}: expected.target must be the fault target {fault_target}, "
            f"got {scenario.expected.target}"
        )
    return scenario


def load_scenarios(names: Sequence[str] = (), directory: Path = SCENARIO_DIR) -> list[Scenario]:
    paths = sorted(directory.glob("*.yaml"))
    available = {path.stem: path for path in paths}
    unknown = [name for name in names if name not in available]
    if unknown:
        raise ScenarioError(
            f"unknown scenario: {', '.join(unknown)} (have {', '.join(available) or 'none'})"
        )
    chosen = [available[name] for name in names] if names else paths
    return [load_scenario(path) for path in chosen]
```

```python
import shutil
import subprocess
import sys
from collections.abc import Sequence
from pathlib import Path

from pydantic import BaseModel, ValidationError

AGENT_COMMAND = "hoa"
AGENT_TIMEOUT_SECONDS = 900
# hoa investigate의 종료 코드 — 2는 설정 오류, 3은 LLM 요청 한도
SETUP_ERROR_EXIT = 2
RATE_LIMITED_EXIT = 3


class AgentNotFound(Exception):
    pass


class ReportedEvidence(BaseModel):
    tool: str
    quote: str


class ReportedDiagnosis(BaseModel):
    cause_label: str
    target: str
    evidence: list[ReportedEvidence]
    confidence: float
    summary_ko: str


class ReportedStep(BaseModel):
    tool: str
    args: dict[str, object]
    output: str
    error: bool
    executed: bool = True


class ReportedUsage(BaseModel):
    model_calls: int
    input_tokens: int
    output_tokens: int
    total_tokens: int


class AgentReport(BaseModel):
    """hoa investigate --json 출력 — 멤버 경계라 hoa_agent 모델을 import하지 않고 따로 둔다."""

    model: str
    alert: str
    diagnosis: ReportedDiagnosis
    stop_reason: str
    steps: list[ReportedStep]
    invalid_tool_calls: int
    usage: ReportedUsage
    elapsed_seconds: float


class AgentOutcome(BaseModel):
    report: AgentReport | None = None
    error: str | None = None
    rate_limited: bool = False
    setup_error: bool = False


def agent_command() -> list[str]:
    found = shutil.which(AGENT_COMMAND, path=str(Path(sys.executable).parent)) or shutil.which(
        AGENT_COMMAND
    )
    if found is None:
        raise AgentNotFound(f"{AGENT_COMMAND} not found — run uv sync --locked")
    return [found]


def last_line(text: str) -> str:
    lines = [line for line in text.strip().splitlines() if line.strip()]
    return lines[-1] if lines else "no output"


def schema_problem(exc: ValidationError) -> str:
    first = exc.errors()[0]
    where = ".".join(str(part) for part in first["loc"]) or "output"
    return f"agent output broke the report schema at {where}: {first['msg']}"


def run_agent(alert: str, model: str, *, command: Sequence[str] | None = None) -> AgentOutcome:
    """에이전트를 별도 프로세스로 돌린다 — 평가가 에이전트 코드를 import하지 않는다."""
    argv = [*(command or agent_command()), "investigate", "--json", "--model", model, alert]
    try:
        result = subprocess.run(
            argv,
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=AGENT_TIMEOUT_SECONDS,
            check=False,
        )
    except subprocess.TimeoutExpired:
        return AgentOutcome(error=f"agent timed out after {AGENT_TIMEOUT_SECONDS}s")
    if result.returncode != 0:
        return AgentOutcome(
            error=last_line(result.stderr),
            rate_limited=result.returncode == RATE_LIMITED_EXIT,
            setup_error=result.returncode == SETUP_ERROR_EXIT,
        )
    try:
        return AgentOutcome(report=AgentReport.model_validate_json(result.stdout))
    except ValidationError as exc:
        return AgentOutcome(error=schema_problem(exc))
```

```python
import json
import subprocess
import time
from collections.abc import Callable

from pydantic import BaseModel

from hoa_eval.scenario import DEMO_NAMESPACE, REPO_ROOT, Fault, Symptom

CONTEXT = "kind-hoa"
DEMO_MANIFESTS = REPO_ROOT / "deploy" / "kind" / "demo"
DEMO_DEPLOYMENTS = ("inventory", "orders", "gateway")
KUBECTL_TIMEOUT_SECONDS = 240
POLL_SECONDS = 3.0
SETTLE_TIMEOUT_SECONDS = 120.0


class ClusterError(Exception):
    pass


class SymptomTimeout(Exception):
    pass


class ContainerStatus(BaseModel):
    restart_count: int
    waiting_reason: str | None


def container_statuses(pods_json: str) -> list[ContainerStatus]:
    pods = json.loads(pods_json)
    return [
        ContainerStatus(
            restart_count=status.get("restartCount", 0),
            waiting_reason=status.get("state", {}).get("waiting", {}).get("reason"),
        )
        for pod in pods.get("items", [])
        for status in pod.get("status", {}).get("containerStatuses") or []
    ]


def terminating_pods(pods_json: str) -> list[str]:
    pods = json.loads(pods_json).get("items", [])
    return [pod["metadata"]["name"] for pod in pods if pod["metadata"].get("deletionTimestamp")]


def symptom_met(statuses: list[ContainerStatus], symptom: Symptom) -> bool:
    if symptom.waiting_reasons and any(
        status.waiting_reason in symptom.waiting_reasons for status in statuses
    ):
        return True
    return symptom.min_restarts is not None and any(
        status.restart_count >= symptom.min_restarts for status in statuses
    )


class Kubectl:
    """관리자 컨텍스트를 늘 명시한다 — 개발 PC의 다른 클러스터를 건드리지 않게."""

    def __init__(self, context: str = CONTEXT) -> None:
        self.context = context

    def __call__(self, *args: str) -> str:
        try:
            result = subprocess.run(
                ["kubectl", "--context", self.context, *args],
                capture_output=True,
                text=True,
                encoding="utf-8",
                errors="replace",
                timeout=KUBECTL_TIMEOUT_SECONDS,
                check=False,
            )
        except (subprocess.TimeoutExpired, FileNotFoundError) as exc:
            raise ClusterError(f"kubectl {' '.join(args)} failed: {exc}") from exc
        if result.returncode != 0:
            raise ClusterError(f"kubectl {' '.join(args)} failed: {result.stderr.strip()}")
        return result.stdout


class DemoCluster:
    def __init__(
        self,
        kubectl: Callable[..., str],
        *,
        sleep: Callable[[float], None] = time.sleep,
        clock: Callable[[], float] = time.monotonic,
    ) -> None:
        self.kubectl = kubectl
        self.sleep = sleep
        self.clock = clock

    def inject(self, fault: Fault) -> None:
        self.kubectl(
            "-n",
            fault.namespace,
            "patch",
            f"deployment/{fault.deployment}",
            "--type",
            "strategic",
            "--patch-file",
            str(REPO_ROOT / fault.patch),
        )

    def reset(self) -> None:
        self.kubectl("apply", "-f", str(DEMO_MANIFESTS))
        # 진행 기한을 넘긴 장애도 되돌리도록 새 ReplicaSet으로 다시 시작한다(dev-cluster.sh reset)
        self.kubectl("-n", DEMO_NAMESPACE, "rollout", "restart", "deployment")
        for name in DEMO_DEPLOYMENTS:
            self.kubectl(
                "-n", DEMO_NAMESPACE, "rollout", "status", f"deployment/{name}", "--timeout=180s"
            )
        self.wait_settled()
        # 이전 회차의 이벤트(기본 1시간 보관)가 다음 조사의 get_events에 섞이지 않게
        self.kubectl("-n", DEMO_NAMESPACE, "delete", "events", "--all")

    def wait_settled(self) -> None:
        """rollout status는 새 파드만 본다 — 종료 중인 옛 파드가 남으면 다음 회차 조사에 섞인다."""
        started = self.clock()
        while True:
            left = terminating_pods(self.kubectl("-n", DEMO_NAMESPACE, "get", "pods", "-o", "json"))
            if not left:
                return
            if self.clock() - started >= SETTLE_TIMEOUT_SECONDS:
                raise ClusterError(f"pods still terminating after reset: {', '.join(left)}")
            self.sleep(POLL_SECONDS)

    def wait_for(self, fault: Fault, symptom: Symptom) -> float:
        started = self.clock()
        while True:
            pods = self.kubectl(
                "-n", fault.namespace, "get", "pods", "-l", f"app={fault.deployment}", "-o", "json"
            )
            if symptom_met(container_statuses(pods), symptom):
                return round(self.clock() - started, 1)
            if self.clock() - started >= symptom.timeout_seconds:
                raise SymptomTimeout(
                    f"deployment/{fault.deployment} showed no symptom "
                    f"within {symptom.timeout_seconds}s"
                )
            self.sleep(POLL_SECONDS)
```

```python
import re
from collections.abc import Iterable

from pydantic import BaseModel

from hoa_eval.agent import AgentReport
from hoa_eval.scenario import Expected

# JSON 출력의 이스케이프(\")와 `키: "값"` 꼴 인용을 같은 문장으로 본다
IGNORED_CHARS = str.maketrans("", "", "\\\"'`")
# 이보다 짧은 인용(`Failed`·`Running`·숫자 하나)은 어디에나 있어 근거로 치지 않는다
MIN_QUOTE_CHARS = 8


class Score(BaseModel):
    label_correct: bool
    target_correct: bool
    quotes: int
    quotes_found: int
    tool_calls: int
    tool_errors: int
    invalid_tool_calls: int
    model_calls: int
    input_tokens: int
    output_tokens: int
    elapsed_seconds: float
    stop_reason: str


def normalize(text: str) -> str:
    unescaped = re.sub(r"\\[nrt]", " ", text)
    return " ".join(unescaped.translate(IGNORED_CHARS).split())


def quote_found(quote: str, outputs: Iterable[str]) -> bool:
    needle = normalize(quote).rstrip(".")
    if len(needle) < MIN_QUOTE_CHARS:
        return False
    # 단어 중간에서 시작하거나 끝나는 일치(`restarts: 1` ⊂ `restarts: 12`)는 인용으로 치지 않는다
    pattern = re.compile(
        rf"(?<![0-9A-Za-z_])(?<![0-9A-Za-z_][.\-]){re.escape(needle)}"
        rf"(?![0-9A-Za-z_]|[.\-][0-9A-Za-z_])"
    )
    return any(pattern.search(normalize(output)) for output in outputs)


def score(expected: Expected, report: AgentReport) -> Score:
    executed = [step for step in report.steps if step.executed]
    outputs = [step.output for step in executed if not step.error]
    quotes = [item.quote for item in report.diagnosis.evidence]
    return Score(
        label_correct=report.diagnosis.cause_label == expected.cause_label,
        target_correct=report.diagnosis.target.strip().lower() == expected.target,
        quotes=len(quotes),
        quotes_found=sum(quote_found(quote, outputs) for quote in quotes),
        tool_calls=len(executed),
        tool_errors=sum(step.error for step in executed),
        invalid_tool_calls=report.invalid_tool_calls,
        model_calls=report.usage.model_calls,
        input_tokens=report.usage.input_tokens,
        output_tokens=report.usage.output_tokens,
        elapsed_seconds=report.elapsed_seconds,
        stop_reason=report.stop_reason,
    )
```

```python
import time
from collections.abc import Callable, Sequence
from datetime import UTC, datetime
from typing import Literal, Protocol

from pydantic import BaseModel

from hoa_eval.agent import AgentOutcome, AgentReport
from hoa_eval.cluster import ClusterError, SymptomTimeout
from hoa_eval.scenario import Fault, Scenario, Symptom
from hoa_eval.scoring import Score, score

Agent = Callable[[str, str], AgentOutcome]
# rate_limit·setup·no_symptom은 모델이 답할 기회가 없던 회차, agent는 답을 내지 못한 회차
Failure = Literal["rate_limit", "setup", "no_symptom", "agent"]
STOPPING_FAILURES: tuple[Failure, ...] = ("rate_limit", "setup")


class Cluster(Protocol):
    def inject(self, fault: Fault) -> None: ...

    def reset(self) -> None: ...

    def wait_for(self, fault: Fault, symptom: Symptom) -> float: ...


class RunRecord(BaseModel):
    scenario: str
    model: str
    attempt: int
    started_at: datetime
    symptom_seconds: float | None = None
    model_id: str | None = None
    score: Score | None = None
    error: str | None = None
    failure: Failure | None = None
    reset_error: str | None = None
    report: AgentReport | None = None


def failure_of(outcome: AgentOutcome) -> Failure | None:
    if outcome.rate_limited:
        return "rate_limit"
    if outcome.setup_error:
        return "setup"
    return "agent" if outcome.report is None else None


def run_once(
    scenario: Scenario, model: str, attempt: int, cluster: Cluster, agent: Agent
) -> RunRecord:
    record = RunRecord(
        scenario=scenario.name, model=model, attempt=attempt, started_at=datetime.now(UTC)
    )
    outcome: AgentOutcome | None = None
    try:
        cluster.inject(scenario.fault)
        try:
            record.symptom_seconds = cluster.wait_for(scenario.fault, scenario.wait)
        except SymptomTimeout as exc:
            record.error = str(exc)
            record.failure = "no_symptom"
        else:
            outcome = agent(scenario.alert, model)
    finally:
        try:
            cluster.reset()
        except ClusterError as exc:
            # 이미 끝낸 조사(무료 등급 요청을 쓴 결과)를 잃지 않게 기록에 남기고 배치에서 멈춘다
            record.reset_error = str(exc)
    if outcome is not None:
        record.error = outcome.error
        record.failure = failure_of(outcome)
        if outcome.report is not None:
            record.report = outcome.report
            record.model_id = outcome.report.model
            record.score = score(scenario.expected, outcome.report)
    return record


def run_batch(
    scenarios: Sequence[Scenario],
    model: str,
    repeat: int,
    cluster: Cluster,
    agent: Agent,
    *,
    pause_seconds: float,
    sink: Callable[[RunRecord], None],
    sleep: Callable[[float], None] = time.sleep,
) -> list[RunRecord]:
    """순서대로 하나씩 돈다 — 요청 한도·설정 오류면 남은 회차도 같은 이유로 실패하므로 멈춘다."""
    cluster.reset()
    records: list[RunRecord] = []
    for attempt in range(1, repeat + 1):
        for scenario in scenarios:
            if records:
                sleep(pause_seconds)
            record = run_once(scenario, model, attempt, cluster, agent)
            sink(record)
            records.append(record)
            if record.reset_error is not None:
                raise ClusterError(record.reset_error)
            if record.failure in STOPPING_FAILURES:
                return records
    return records
```

```python
from pathlib import Path

def ratio(hits: int, total: int) -> str:

def short_error(text: str) -> str:
    """완료노트(공개 저장소)에 옮길 표다 — 홈 경로를 가리고, 표를 깨는 `|`와 긴 원문을 줄인다."""
    home = str(Path.home())
    for form in (home.replace("\\", "\\\\"), home, home.replace("\\", "/")):
        text = text.replace(form, "~")
    text = " ".join(text.split()).replace("|", "\\|")
    return text if len(text) <= MAX_ERROR_CHARS else text[: MAX_ERROR_CHARS - 1] + "…"


def model_names(group: Sequence[RunRecord], profile: str) -> str:
    names = list(dict.fromkeys(r.model_id for r in group if r.model_id))
    return ", ".join(names) or profile


def excluded_note(group: Sequence[RunRecord]) -> str:
    counts = Counter(EXCLUDED[r.failure] for r in group if r.failure in EXCLUDED)
    return ", ".join(f"{label} {count}" for label, count in counts.items()) or "0"


def summary_table(records: Sequence[RunRecord]) -> str:
    rows = [
        "| 시나리오 | 모델 | 채점 회차 | 원인 정답 | 대상 정답 | 인용 확인 | 도구 호출 평균 "
        "| 도구 오류 | 형식 깨진 호출 | 시간 평균(초) | 입력 토큰 평균 | 에이전트 실패 | 제외 |",
        "|---|---|---|---|---|---|---|---|---|---|---|---|---|",
    ]
    groups: dict[tuple[str, str], list[RunRecord]] = {}
    for record in records:
        groups.setdefault((record.scenario, record.model), []).append(record)
    for (scenario, profile), group in groups.items():
        judged = [r for r in group if r.failure not in EXCLUDED]
        scored = [r.score for r in judged if r.score is not None]
        head = f"| {scenario} | {model_names(group, profile)} | {len(judged)} "
        tail = f"| {len(judged) - len(scored)} | {excluded_note(group)} |"
        if not scored:
            rows.append(f"{head}| 0/{len(judged)} | 0/{len(judged)} | - | - | - | - | - | - {tail}")
            continue
        rows.append(
            f"{head}"
            f"| {ratio(sum(s.label_correct for s in scored), len(judged))} "
            f"| {ratio(sum(s.target_correct for s in scored), len(judged))} "
            f"| {ratio(sum(s.quotes_found for s in scored), sum(s.quotes for s in scored))} "
            f"| {mean(s.tool_calls for s in scored):.1f} "
            f"| {sum(s.tool_errors for s in scored)} "
            f"| {sum(s.invalid_tool_calls for s in scored)} "
            f"| {mean(s.elapsed_seconds for s in scored):.1f} "
            f"| {mean(s.input_tokens for s in scored):,.0f} "
            f"{tail}"
        )
    return "\n".join(rows)


def runs_table(records: Sequence[RunRecord]) -> str:
    rows = [
        "| 시나리오 | 모델 | 회차 | 원인 | 인용 | 도구 | 도구 오류 | 시간(초) | 멈춘 이유 | 오류 |",
        "|---|---|---|---|---|---|---|---|---|---|",
    ]
    for record in records:
        name = record.model_id or record.model
        if record.score is None or record.report is None:
            rows.append(
                f"| {record.scenario} | {name} | {record.attempt} | - | - | - | - | - | - "
                f"| {short_error(record.error or record.reset_error or '-')} |"
            )
            continue
        s = record.score
        rows.append(
            f"| {record.scenario} | {name} | {record.attempt} "
            f"| {record.report.diagnosis.cause_label}{'' if s.label_correct else ' ✗'} "
            f"| {ratio(s.quotes_found, s.quotes)} | {s.tool_calls} | {s.tool_errors} "
            f"| {s.elapsed_seconds:.1f} | {s.stop_reason} "
            f"| {short_error(record.reset_error) if record.reset_error else '-'} |"
        )
    return "\n".join(rows)
```

```python
def non_negative_float(text: str) -> float:

def build_parser() -> argparse.ArgumentParser:

def use_utf8_output() -> None:
    """Windows에서 출력을 파일·파이프로 돌리면 로캘 인코딩(cp949)이 돼 도움말·표의 문자가 깨진다."""
    for stream in (sys.stdout, sys.stderr):
        if isinstance(stream, io.TextIOWrapper):
            stream.reconfigure(encoding="utf-8")


def shown(path: Path) -> str:
    """공개 문서에 옮길 수 있게 저장소 기준 경로로 보여 준다."""
    try:
        return path.resolve().relative_to(REPO_ROOT).as_posix()
    except ValueError:
        return str(path)


def expand(patterns: Sequence[str]) -> list[Path]:
    """PowerShell·cmd는 와일드카드를 펼쳐 주지 않는다 — 여기서 펼친다."""
    paths: list[Path] = []
    for pattern in patterns:
        matches = sorted(glob.glob(pattern))
        if matches:
            paths.extend(Path(match) for match in matches)
        else:
            paths.append(Path(pattern))
    return paths


def run_command(args: argparse.Namespace, cluster: Cluster, agent: Agent) -> int:
    try:
        scenarios = load_scenarios(args.scenario)
    except ScenarioError as exc:
        print(f"hoa-eval: {exc}", file=sys.stderr)
        return 2
    args.results_dir.mkdir(parents=True, exist_ok=True)
    stamp = datetime.now(UTC).strftime("%Y%m%dT%H%M%SZ")
    path = args.results_dir / f"{stamp}-{args.model}.jsonl"

    def write(record: RunRecord) -> None:
        with path.open("a", encoding="utf-8") as sink:
            sink.write(record.model_dump_json() + "\n")
        verdict = record.error or (
            "맞음" if record.score and record.score.label_correct else "틀림"
        )
        print(
            f"{record.scenario} #{record.attempt}: {short_error(verdict)}",
            file=sys.stderr,
            flush=True,
        )

    try:
        records = run_batch(
            scenarios, args.model, args.repeat, cluster, agent, pause_seconds=args.pause, sink=write
        )
    except (ClusterError, AgentNotFound) as exc:
        print(f"hoa-eval: {short_error(str(exc))}", file=sys.stderr)
        if path.exists():
            print(f"지금까지의 결과: {shown(path)}", file=sys.stderr)
        return 1
    print(summary_table(records))
    print(f"\n결과: {shown(path)}")
    last = records[-1].failure if records else None
    if last == "setup":
        print("hoa-eval: 에이전트 설정 오류로 남은 회차를 멈췄다", file=sys.stderr)
        return 2
    if last == "rate_limit":
        print("hoa-eval: LLM 요청 한도에 걸려 남은 회차를 멈췄다", file=sys.stderr)
        return 3
    return 0


def report_command(args: argparse.Namespace) -> int:
    try:
        records = load_records(expand(args.files))
    except (OSError, ValidationError) as exc:
        print(f"hoa-eval: cannot read results — {short_error(str(exc))}", file=sys.stderr)
        return 2
    print(summary_table(records))
    print()
    print(runs_table(records))
    return 0


def main(
    argv: Sequence[str] | None = None,
    *,
    cluster: Cluster | None = None,
    agent: Agent = run_agent,
) -> int:
    use_utf8_output()
    parser = build_parser()
    args = parser.parse_args(argv)
    if args.command == "run":
        return run_command(args, cluster or DemoCluster(Kubectl()), agent)
    if args.command == "report":
        return report_command(args)
    parser.print_help()
    return 0
```
