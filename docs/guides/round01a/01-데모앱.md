# round01a 교본 3-1 — 데모 앱(`demo_app`)

> 「00-개요」의 「3. 구현」에 해당한다. 아래 파일을 이 순서대로 그대로 만든다(전체 수록, 발췌 없음).

진단 대상이 되는 작은 3-서비스 앱이다. 호출 흐름은 `gateway → orders → inventory`이고 데이터는 메모리에 둔다. 이미지는 하나이고 역할은 환경변수 `HOA_DEMO_ROLE`로 고른다. 설정이 틀리면 무엇이 틀렸는지 한 줄을 남기고 종료 코드 1로 끝난다 — 장애 `bad-config`의 단서가 이 한 줄이다.

만드는 순서: `pyproject.toml` → 설정(`config.py`) → 모델·공용(`models.py`·`common.py`) → 서비스 셋(`inventory.py`·`orders.py`·`gateway.py`) → 진입점(`cli.py`·`__main__.py`) → 테스트 → 이미지(`Dockerfile`·`Dockerfile.dockerignore`). 테스트는 TDD 순서(실패 확인 → 구현 → 통과)로 서비스마다 먼저 쓴다. 앱은 httpx `ASGITransport`로 서버 없이 부른다.

**파일**: `demo_app/src/hoa_demo/config.py`

```python
from collections.abc import Mapping
from enum import StrEnum

from pydantic import BaseModel, Field, ValidationError, field_validator


class Role(StrEnum):
    GATEWAY = "gateway"
    ORDERS = "orders"
    INVENTORY = "inventory"


class DemoSettings(BaseModel):
    role: Role
    port: int = Field(ge=1, le=65535)
    orders_url: str | None = None
    inventory_url: str | None = None

    @field_validator("orders_url", "inventory_url")
    @classmethod
    def http_url_without_trailing_slash(cls, value: str | None) -> str | None:
        if value is None:
            return None
        if not value.startswith(("http://", "https://")):
            raise ValueError("must start with http:// or https://")
        return value.rstrip("/")


class ConfigError(Exception):
    pass


ENV_NAMES = {
    "role": "HOA_DEMO_ROLE",
    "port": "HOA_DEMO_PORT",
    "orders_url": "HOA_DEMO_ORDERS_URL",
    "inventory_url": "HOA_DEMO_INVENTORY_URL",
}


def load_settings(env: Mapping[str, str]) -> DemoSettings:
    raw = {field: env[name] for field, name in ENV_NAMES.items() if name in env}
    try:
        return DemoSettings.model_validate(raw)
    except ValidationError as exc:
        error = exc.errors()[0]
        field = str(error["loc"][0])
        raise ConfigError(f"{ENV_NAMES[field]}={raw.get(field)!r}: {error['msg']}") from exc


def require(value: str | None, field: str, role: Role) -> str:
    if not value:
        raise ConfigError(f"{ENV_NAMES[field]} is required for role {role}")
    return value
```

import httpx
from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, ValidationError

from hoa_demo.models import ErrorBody, Health


def add_health(app: FastAPI) -> None:
    @app.get("/healthz")
    async def healthz() -> Health:
        return Health()


def downstream_detail(response: httpx.Response) -> str:
    try:
        return ErrorBody.model_validate_json(response.content).detail
    except ValidationError:
        return ""


async def fetch[M: BaseModel](
    model: type[M],
    url: str,
    *,
    service: str,
    timeout: float,
    transport: httpx.AsyncBaseTransport | None = None,
) -> M:
    try:
        async with httpx.AsyncClient(transport=transport, timeout=timeout) as client:
            response = await client.get(url)
    except httpx.HTTPError as exc:
        raise HTTPException(
            status_code=502, detail=f"{service} unreachable: {type(exc).__name__}"
        ) from exc
    detail = downstream_detail(response)
    if response.status_code == 404:
        raise HTTPException(status_code=404, detail=f"{service}: {detail or 'not found'}")
    if response.status_code != 200:
        suffix = f": {detail}" if detail else ""
        raise HTTPException(
            status_code=502, detail=f"{service} returned {response.status_code}{suffix}"
        )
    try:
        return model.model_validate_json(response.content)
    except ValidationError as exc:
        raise HTTPException(status_code=502, detail=f"{service} returned an invalid body") from exc
```

import httpx
from fastapi import FastAPI, HTTPException

from hoa_demo.common import add_health, fetch
from hoa_demo.models import Order, OrderDetail, Stock

INVENTORY_TIMEOUT_SECONDS = 2.0

ORDERS = {
    1: Order(order_id=1, sku="SKU-1", quantity=2),
    2: Order(order_id=2, sku="SKU-2", quantity=1),
}


def create_app(inventory_url: str, transport: httpx.AsyncBaseTransport | None = None) -> FastAPI:
    app = FastAPI(title="hoa-demo orders")
    add_health(app)

    @app.get("/orders/{order_id}")
    async def get_order(order_id: int) -> OrderDetail:
        order = ORDERS.get(order_id)
        if order is None:
            raise HTTPException(status_code=404, detail=f"unknown order: {order_id}")
        stock = await fetch(
            Stock,
            f"{inventory_url}/stock/{order.sku}",
            service="inventory",
            timeout=INVENTORY_TIMEOUT_SECONDS,
            transport=transport,
        )
        return OrderDetail(order=order, stock=stock)

    return app
```

import httpx
from fastapi import FastAPI

from hoa_demo.common import add_health, fetch
from hoa_demo.models import OrderDetail

# orders가 inventory를 기다리는 시간보다 길게 둔다 —
# inventory가 멈췄을 때 원인이 orders로 잘못 보이지 않게

def create_app(orders_url: str, transport: httpx.AsyncBaseTransport | None = None) -> FastAPI:

    add_health(app)

    @app.get("/api/orders/{order_id}")
    async def get_order(order_id: int) -> OrderDetail:
        return await fetch(
            OrderDetail,
            f"{orders_url}/orders/{order_id}",
            service="orders",
            timeout=ORDERS_TIMEOUT_SECONDS,
            transport=transport,
        )

    return app

## 3.9 `demo_app/src/hoa_demo/cli.py`

**파일**: `demo_app/src/hoa_demo/cli.py`

```python
import logging
import os
import sys

import uvicorn
from fastapi import FastAPI

from hoa_demo import gateway, inventory, orders
from hoa_demo.config import ConfigError, DemoSettings, Role, load_settings, require


class SkipHealthChecks(logging.Filter):
    """Keeps probe hits out of the access log so they do not push real clues out of the tail."""

    def filter(self, record: logging.LogRecord) -> bool:
        return "/healthz" not in record.getMessage()


def build_app(settings: DemoSettings) -> FastAPI:
    if settings.role is Role.INVENTORY:
        return inventory.create_app()
    if settings.role is Role.ORDERS:
        return orders.create_app(require(settings.inventory_url, "inventory_url", settings.role))
    return gateway.create_app(require(settings.orders_url, "orders_url", settings.role))


def main() -> None:
    try:
        settings = load_settings(os.environ)
        app = build_app(settings)
    except ConfigError as exc:
        print(f"hoa-demo: invalid configuration: {exc}", file=sys.stderr, flush=True)
        raise SystemExit(1) from exc
    logging.getLogger("uvicorn.access").addFilter(SkipHealthChecks())
    uvicorn.run(app, host="0.0.0.0", port=settings.port, log_level="info")
```
